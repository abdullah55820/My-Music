from pathlib import Path import zipfile, re img = Path("/mnt/data/FB2DA2E5-DAA5-446E-A7B1-1DAB47BD43A7.jpeg") base = Path("/mnt/data/index.html") html = base.read_text(encoding="utf-8") # Prevent YouTube thumbnail from becoming the page background. html = re.sub( r'\s*document\.getElementById\(\s*"bg"\s*\)\.style\.backgroundImage\s*=\s*[^\n;]+;?', '', html ) # Ensure the background is the supplied image with no blur. html = re.sub( r'\.bg\{.*?\}', '''.bg{ position:fixed; inset:0; background:url("background.jpg") center center/cover no-repeat; transform:scale(1.01); }''', html, count=1, flags=re.S ) html = html.replace('filter:blur(35px) brightness(0.65);', '') html = html.replace('filter: blur(35px) brightness(0.65);', '') out = Path("/mnt/data/BD_Live_Final") out.mkdir(exist_ok=True) (out / "index.html").write_text(html, encoding="utf-8") (out / "background.jpg").write_bytes(img.read_bytes()) zip_path = Path("/mnt/data/BD_Live_Final.zip") with zipfile.ZipFile(zip_path, "w", zipfile.ZIP_DEFLATED) as z: z.write(out / "index.html", "index.html") z.write(out / "background.jpg", "background.jpg") print(zip_path)
